# 解题思路

这道题询问能否将数组 AAA 重新排序成数组 BBB 使得 BBB 是个等比数列。 我们考虑等比数列的性质:

  1. 如果比值 rrr 是正数，那么肯定看 AAA 升序或者降序排序后是否能成为一个等比数列。
  2. 但是比值 rrr 为负数，那么这时候对于 AAA 的排序应该修改。比如 当 Ai=1,r=−2A_i = 1, r = -2A​i​​=1,r=−2 时，A2=−2,A3=4,A4=−8A_2 = -2, A_3 = 4, A_4 = -8A​2​​=−2,A​3​​=4,A​4​​=−8 那么此时使用升序排序：A'={-8，-2，1，4} 此时就不是等比数列。那么我们看到原数组 AAA，注意到数组 B=∣Ai∣B={|A_i|}B=∣A​i​​∣ 是个单调队列，所以我们可以考虑通过按照绝对值排序 AAA。如果 A=(−1)i,i∈randomA={(-1)^i, i \in random}A=(−1)​i​​,i∈random 呢，此时排序后和原来输入的完全一致无法判断是否成立，那么我们继续考虑：
    1. 如果 AAA 数组中既有正数也有负数，那么等比数列的排序必然是一个正数前后都有一个负数，所以 ∣x−y∣≤1|x-y|\leq 1∣x−y∣≤1（其中 xxx 为正数个数, yyy 为负数个数），输入后特判即可。

# 解题代码

//Atcoder板子
#include <bits/stdc++.h>
using namespace std;
int T,N,A[200005];
bool cmp (const int& a, const int& b) {return abs(a) < abs(b);}
#include <utility>
#include <algorithm>

using namespace std;

// 计算最大公约数
int gcd(int a, int b) {
    return b == 0 ? a : gcd(b, a % b);
}

pair<int, int> simplifyFraction(pair<int, int> fraction) {
    int p = fraction.first;
    int q = fraction.second;
    // 计算最大公约数
    int commonDivisor = gcd(abs(p), abs(q));
    
    // 化简分数
    p /= commonDivisor;
    q /= commonDivisor;
    
    // 确保分母始终为正
    if (q < 0) {
        p = -p;
        q = -q;
    }
    
    return make_pair(p, q);
}

void work(){
    cin>>N;
    int x=0,y=0;
    for(int i=1;i<=N;i++){
        cin>>A[i];
        if(A[i]>0)x++;
        else y++;
    }
    if(x&&y&&abs(x-y)>1){cout<<"No"<<endl;return;}
    sort(A+1,A+N+1,cmp);
    if(A[1]==0){cout<<"No"<<endl;return;}
    bool is=false;
    for(int i=1;i<N;i++)if(abs(A[i])!=abs(A[i+1]))is=true;
    if(!is){
        cout<<"Yes"<<endl;
        return;
    }
    pair<int, int> fraction = make_pair(A[2], A[1]);
    fraction = simplifyFraction(fraction);
    for(int i=2;i<N;i++){
        if(A[i]*fraction.first!=A[i+1]*fraction.second){
            cout<<"No"<<endl;
            return;
        }
    }
    cout<<"Yes"<<endl;
    return;
}
int main() {
    cin>>T;
    while(T--)work();
    return 0;
}
